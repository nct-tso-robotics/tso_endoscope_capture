cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(endoscope_capture_backend_tests LANGUAGES CXX)
  option(CAPTURE_BUILD_ROS "Build ROS components through the repository root" OFF)
  if(CAPTURE_BUILD_ROS)
    message(FATAL_ERROR "Configure the repository root to build the ROS package")
  endif()
endif()

if(CAPTURE_ROS_VERSION STREQUAL "2")
  set(CMAKE_CXX_STANDARD 17)
else()
  set(CMAKE_CXX_STANDARD 14)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Build type" FORCE)
endif()
option(CAPTURE_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(CAPTURE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(decklink_dispatch STATIC vendor/decklink/DeckLinkAPIDispatch.cpp)
target_include_directories(decklink_dispatch SYSTEM PUBLIC vendor/decklink)
target_link_libraries(decklink_dispatch PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(endoscope_capture_backend STATIC src/decklink_capture.cpp)
target_include_directories(endoscope_capture_backend PUBLIC include)
target_link_libraries(endoscope_capture_backend PUBLIC decklink_dispatch Threads::Threads)
target_compile_options(endoscope_capture_backend PRIVATE -Wall -Wextra -Wpedantic)

if(CAPTURE_BUILD_ROS)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc)
  find_package(yaml-cpp REQUIRED CONFIG)
  if(TARGET yaml-cpp::yaml-cpp)
    set(capture_yaml_target yaml-cpp::yaml-cpp)
  elseif(TARGET yaml-cpp)
    set(capture_yaml_target yaml-cpp)
  else()
    message(FATAL_ERROR "The yaml-cpp package must export either yaml-cpp::yaml-cpp or yaml-cpp; check the selected yaml-cpp CMake configuration")
  endif()
  add_library(endoscope_stereo_processor STATIC src/stereo_processor.cpp)
  target_include_directories(endoscope_stereo_processor PUBLIC include)
  target_include_directories(endoscope_stereo_processor SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(endoscope_stereo_processor PUBLIC ${OpenCV_LIBS} ${capture_yaml_target})
  target_compile_options(endoscope_stereo_processor PRIVATE -Wall -Wextra -Wpedantic)
  if(CAPTURE_ROS_VERSION STREQUAL "2")
    target_compile_definitions(endoscope_stereo_processor PUBLIC CAPTURE_ROS2)
    ament_target_dependencies(endoscope_stereo_processor PUBLIC sensor_msgs std_msgs)
    add_library(endoscope_ros2_publisher STATIC src/ros2_publisher.cpp)
    target_link_libraries(endoscope_ros2_publisher PUBLIC endoscope_stereo_processor
      "${capture_typesupport_target}")
    ament_target_dependencies(endoscope_ros2_publisher PUBLIC rclcpp sensor_msgs std_msgs)
    target_compile_options(endoscope_ros2_publisher PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(endoscope_capture_node src/ros2_capture_node.cpp)
    target_link_libraries(endoscope_capture_node PRIVATE endoscope_capture_backend
      endoscope_ros2_publisher)
    ament_target_dependencies(endoscope_capture_node PUBLIC camera_info_manager rcl_interfaces)
    # ROS1 searches the enclosing source package recursively for executables.
    # Hide the ROS2 build copy so the two versions keep unambiguous discovery.
    set_target_properties(endoscope_capture_node PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/.bin")
    install(TARGETS endoscope_capture_node RUNTIME DESTINATION lib/${PROJECT_NAME})
  else()
    target_include_directories(endoscope_stereo_processor SYSTEM PUBLIC ${capture_ros_includes})
    target_link_libraries(endoscope_stereo_processor PUBLIC ${capture_ros_libraries})
    add_library(endoscope_stereo_splitter STATIC src/stereo_splitter.cpp)
    target_link_libraries(endoscope_stereo_splitter PUBLIC endoscope_stereo_processor)
    target_compile_options(endoscope_stereo_splitter PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(endoscope_capture_node src/ros_capture_node.cpp)
    target_include_directories(endoscope_capture_node SYSTEM PRIVATE ${CATKIN_DEVEL_PREFIX}/include)
    target_link_libraries(endoscope_capture_node PRIVATE endoscope_capture_backend endoscope_stereo_splitter)
    add_dependencies(endoscope_capture_node ${${PROJECT_NAME}_EXPORTED_TARGETS} ${catkin_EXPORTED_TARGETS})
    set_target_properties(endoscope_capture_node PROPERTIES
      RUNTIME_OUTPUT_DIRECTORY "${CATKIN_DEVEL_PREFIX}/${CATKIN_PACKAGE_BIN_DESTINATION}")
    install(TARGETS endoscope_capture_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  endif()
  target_compile_options(endoscope_capture_node PRIVATE -Wall -Wextra -Wpedantic)
endif()
